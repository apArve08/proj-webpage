import torch
import torch.nn as nn
import math

# AP Tensor Lab: AxonPerceptron + Self-Attention
class ScaledDotProductAttention(nn.Module):
    def forward(self, Q, K, V, tau=1.0, mask=None):
        d_k = Q.size(-1)
        scores = (Q @ K.transpose(-2, -1)) / (math.sqrt(d_k) * tau)
        if mask is not None:
            scores = scores.masked_fill(mask == 0, -1e9)
        weights = torch.softmax(scores, dim=-1)
        return weights @ V, weights

class AxonPerceptronBlock(nn.Module):
    def __init__(self, d_model=512):
        super().__init__()
        self.attn = ScaledDotProductAttention()
        self.ffn = nn.Sequential(nn.Linear(d_model, d_model), nn.GELU())

The Axon

In neurobiology, the axon is the transmission line of a biological neuron that conducts electrical impulses away from the cell body to connect with other neurons. It represents natural intuition, sensory input, and organic connectivity.

The Perceptron

Invented by Frank Rosenblatt in 1957, the perceptron is the foundational mathematical unit of artificial neural networks. By applying weights, biases, and activation thresholds to multi-dimensional tensors, raw signals become machine intelligence.
